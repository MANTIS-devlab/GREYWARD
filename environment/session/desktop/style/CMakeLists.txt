cmake_minimum_required(VERSION 3.20)
project(greyward_desktop_style LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(greywarddesktop MODULE obsidian.cpp)
target_link_libraries(greywarddesktop PRIVATE Qt6::Widgets)
target_compile_options(greywarddesktop PRIVATE -Wall -Wextra -Werror)
install(TARGETS greywarddesktop LIBRARY DESTINATION lib/greyward/desktop/qt/styles)
